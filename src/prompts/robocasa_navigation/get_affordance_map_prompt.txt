import numpy as np
from perception_utils import parse_query_obj
from plan_utils import get_empty_affordance_map, set_pixel_by_radius, world_offset

# How to set affordance for the navigation goal:
#  - Default radius: 30 cm  (smaller would create degenerate single-pixel
#    targets; A* / greedy planner needs a region).
#  - For "near object X" queries:   pass the parsed object directly.
#       set_pixel_by_radius(affordance_map, X, radius_cm=30, value=1)
#  - For "X cm in front of / next to / etc." queries: use world_offset(...)
#    which returns a point in WORLD-frame metres so the planner grid
#    conversion is correct on every layout (avoids the per-layout
#    100³ vs 162×162 mismatch in raw .position arithmetic).
#       pt = world_offset(X, dx_m=0.15, dy_m=0)   # 15cm in +x world
#       set_pixel_by_radius(affordance_map, pt, radius_cm=30, value=1)
#
# Axis convention (robocasa kitchen, world frame):
#       +x → counter-front direction
#       +y → robot-left direction
#       +z → up
# "in front of" usually means a small +x offset in the fixture's neighbourhood.
# DO NOT use raw `.position` indices in `set_pixel_by_radius([row, col], ...)`
# — those are in a 100³ legacy grid that does not match the planner's grid
# on rectangular layouts.

# Query: a point 15cm from the coffee machine.
affordance_map = get_empty_affordance_map(task='navigation')
coffee_machine = parse_query_obj('coffee_machine')
# Object mode — radius_cm dilates the real fixture mesh, no offset arithmetic.
set_pixel_by_radius(affordance_map, coffee_machine, radius_cm=30, value=1)
ret_val = affordance_map

# Query: a point 15cm in front of the coffee machine.
affordance_map = get_empty_affordance_map(task='navigation')
coffee_machine = parse_query_obj('coffee_machine')
pt = world_offset(coffee_machine, dx_m=0.15, dy_m=0)
set_pixel_by_radius(affordance_map, pt, radius_cm=30, value=1)
ret_val = affordance_map
