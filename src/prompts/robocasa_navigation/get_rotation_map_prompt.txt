import numpy as np
from plan_utils import get_empty_rotation_map, set_pixel_by_radius, cm2index
from perception_utils import parse_query_obj

# NOTE: For navigation tasks, rotation_map is a 2D float array (shape: [map_size, map_size])
# of yaw angles in radians. Always use get_empty_rotation_map(task='navigation').
# Set scalar yaw values. Do NOT use .normal or .orientation — use .position for direction.

# Query: face toward the door when approaching.
rotation_map = get_empty_rotation_map(task='navigation')
door = parse_query_obj('door')
px, py, _ = door.position
rotation_map[:, :] = np.arctan2(py, px)
ret_val = rotation_map

# Query: face the cat directly with a 90-degree turning angle.
rotation_map = get_empty_rotation_map(task='navigation')
cat = parse_query_obj('cat')
px, py, _ = cat.position
rotation_map[:, :] = np.arctan2(py, px)
ret_val = rotation_map

# Query: turn to face the dog when within 60cm.
rotation_map = get_empty_rotation_map(task='navigation')
dog = parse_query_obj('dog')
px, py, _ = dog.position
target_yaw = np.arctan2(py, px)
set_pixel_by_radius(rotation_map, dog.position, radius_cm=60, value=target_yaw)
ret_val = rotation_map

# Query: no specific rotation requirement.
rotation_map = get_empty_rotation_map(task='navigation')
ret_val = rotation_map

# Query: face the exit path with 45-degree rotation.
rotation_map = get_empty_rotation_map(task='navigation')
rotation_map[:, :] = np.pi / 4
ret_val = rotation_map
