import numpy as np
from perception_utils import parse_query_obj
from action_utils import composer
# NOTE — runtime substitution:
#   The 'objects' lists below use placeholder names: 'goal' for the
#   destination and 'object_1', 'object_2', ... for non-goal items.
#   At runtime the harness replaces these with the actual scene names
#   (e.g. ['human', 'cat', 'sink']). In your composer() string, refer
#   to the goal as 'the goal' and any non-goal item as 'the obstacle'.
#
# CRITICAL — goal vs obstacle distinction:
#   The GOAL is the destination the robot must reach and approach closely.
#   OBSTACLES are everything else along the path that must be avoided.
#   At runtime, the destination's actual name (e.g., 'human', 'fridge')
#   may also appear in the objects list — treat it ONLY as the goal,
#   NEVER as something to keep clearance from. If the instruction says
#   "navigate to the human", then 'human' is the goal: it goes only
#   into "move to the goal" and MUST NOT appear in any "keeping ... cm
#   from ..." clause. Adding an avoidance distance for the goal would
#   block the robot from reaching it.
#
# NOTE — value ranges:
#   distance: minimum 0, maximum 80 cm.
#   velocity: minimum 0, maximum 2 (1 = nominal speed). Express it as a
#   percent in the get_velocity_map query, where 100% = nominal.
#
# NOTE — what to emit:
#   You may write the constraint either with abstract verbs (e.g. "stay
#   clear of the obstacle", "slow down when near it") or with specific
#   cm distances (e.g. "keep at least 50cm from the obstacle"). Mixing is
#   fine. The composer downstream translates both styles into concrete
#   map calls.
#
# NOTE — choose constraints freely but keep them feasible:
#   The composer can pick any value in its allowed range. Use the full
#   vocabulary — from "passing right next to" through "being careful
#   around" to "staying as far as possible". Avoid demanding so much
#   clearance that no feasible path remains for the robot.
#
# NOTE — only mention what the task needs:
#   If there is no obstacle, do not mention avoidance. If nominal speed
#   is fine, do not mention velocity. Combine all constraints for one
#   goal into a single composer() call. Use multiple composer() calls
#   only for multi-stage tasks.

objects = ['goal']
# Query: navigate to the goal.
# Reasoning: no obstacle is mentioned.
composer("move to the goal")
# done

objects = ['goal', 'object_1']
# Query: navigate to the goal while keeping a safe distance from the obstacle.
# Reasoning: the obstacle is solid.
composer("move to the goal while keeping at least 20cm from the obstacle")
# done

objects = ['goal', 'object_1', 'object_2']
# Query: navigate to the goal while being careful around the two obstacles.
# Reasoning: the first obstacle is a moving presence. The second is delicate.
composer("move to the goal while keeping at least 70cm from the first obstacle and at least 30cm from the second obstacle, slowing down when near either of them")
# done

objects = ['goal', 'object_1', 'object_2', 'object_3']

# Reasoning: three obstacles, no room to spare.
composer("move to the goal while staying clear of every obstacle and slowing down significantly throughout the path")
# done

objects = ['goal', 'object_1', 'object_2']
# Query: navigate to the goal carefully — there are two obstacles.
# Reasoning: the first is delicate. The second occupies its space dynamically.
composer("move to the goal while keeping at least 35cm from the first obstacle and slowing down to 30% velocity when near it, and keeping at least 10cm from the second obstacle")
# done
